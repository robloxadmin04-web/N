<script>
(async function () {
  var session = await requireAuth();
  if (!session) return;

  var GID = qs('id');
  if (!GID) {
    window.location.replace('dashboard.html');
    return;
  }

  var BASE = '/api/guilds/' + encodeURIComponent(GID);

  // ---------- tabs ----------
  document.querySelectorAll('.tabs button').forEach(function (btn) {
    btn.addEventListener('click', function () {
      document.querySelectorAll('.tabs button').forEach(function (b) { b.classList.remove('on'); });
      document.querySelectorAll('.tab-body').forEach(function (b) { b.classList.remove('on'); });
      btn.classList.add('on');
      document.getElementById('tab-' + btn.dataset.tab).classList.add('on');
    });
  });

  // ---------- initial load ----------
  var settings, channels, roles;

  try {
    var results = await Promise.all([
      api(BASE + '/settings'),
      api(BASE + '/channels'),
      api(BASE + '/roles')
    ]);
    settings = results[0].settings;
    channels = results[1].channels || [];
    roles = results[2].roles || [];
  } catch (e) {
    toast(e.message, 'err');
    return;
  }

  fillSelect(document.getElementById('welcome_channel_id'), channels, settings.welcome_channel_id, 'Walang napili');
  fillSelect(document.getElementById('log_channel_id'), channels, settings.log_channel_id, 'Walang napili');
  fillSelect(document.getElementById('updates_channel_id'), channels, settings.updates_channel_id, 'Walang napili');
  fillSelect(document.getElementById('status_channel_id'), channels, settings.status_channel_id, 'Walang napili');
  fillSelect(document.getElementById('autorole_id'), roles, settings.autorole_id, 'Walang napili');
  fillSelect(document.getElementById('map_role'), roles, '', 'Pumili ng role');

  document.getElementById('welcome_enabled').checked = !!settings.welcome_enabled;
  document.getElementById('autorole_enabled').checked = !!settings.autorole_enabled;
  document.getElementById('raid_protection').checked = !!settings.raid_protection;
  document.getElementById('welcome_message').value = settings.welcome_message || '';
  document.getElementById('min_account_age_days').value = settings.min_account_age_days || 0;
  document.getElementById('prefix').value = settings.prefix || '!';

  // ---------- save settings ----------
  document.getElementById('saveSettings').addEventListener('click', async function () {
    var btn = this;
    btn.disabled = true;

    var body = {
      prefix: document.getElementById('prefix').value.trim() || '!',
      welcome_enabled: document.getElementById('welcome_enabled').checked,
      welcome_channel_id: document.getElementById('welcome_channel_id').value || null,
      welcome_message: document.getElementById('welcome_message').value,
      autorole_enabled: document.getElementById('autorole_enabled').checked,
      autorole_id: document.getElementById('autorole_id').value || null,
      log_channel_id: document.getElementById('log_channel_id').value || null,
      updates_channel_id: document.getElementById('updates_channel_id').value || null,
      status_channel_id: document.getElementById('status_channel_id').value || null,
      raid_protection: document.getElementById('raid_protection').checked,
      min_account_age_days: Number(document.getElementById('min_account_age_days').value || 0)
    };

    try {
      await api(BASE + '/settings', { method: 'PUT', body: body });
      toast('Na save na ang settings.', 'ok');
    } catch (e) {
      toast(e.message, 'err');
    }
    btn.disabled = false;
  });

  // ---------- premium mapping ----------
  async function loadMappings() {
    var box = document.getElementById('mappings');
    try {
      var out = await api(BASE + '/premium-roles');
      var list = out.mappings || [];

      if (list.length === 0) {
        box.innerHTML = '<div class="empty">Wala pang mapping.</div>';
        return;
      }

      var html = '';
      list.forEach(function (m) {
        var role = roles.find(function (r) { return r.id === m.role_id; });
        html += '<div class="item"><div><div class="t">' + esc(m.tier) + '</div>' +
                '<div class="s">' + esc(role ? role.name : m.role_id) + '</div></div></div>';
      });
      box.innerHTML = html;
    } catch (e) {
      box.innerHTML = '<div class="empty">' + esc(e.message) + '</div>';
    }
  }

  document.getElementById('saveMapping').addEventListener('click', async function () {
    var tier = document.getElementById('map_tier').value.trim();
    var roleId = document.getElementById('map_role').value;

    if (!tier || !roleId) {
      toast('Kailangan ng tier at role.', 'err');
      return;
    }

    try {
      await api(BASE + '/premium-roles', { method: 'PUT', body: { tier: tier, role_id: roleId } });
      toast('Na save ang mapping.', 'ok');
      loadMappings();
    } catch (e) {
      toast(e.message, 'err');
    }
  });

  // ---------- premium members ----------
  async function loadPremium() {
    var box = document.getElementById('premiumList');
    try {
      var out = await api(BASE + '/premium');
      var list = out.members || [];

      if (list.length === 0) {
        box.innerHTML = '<div class="empty">Wala pang premium member.</div>';
        return;
      }

      var html = '';
      list.forEach(function (m) {
        var left = daysLeft(m.expires_at);
        var sub = fmtDate(m.expires_at);
        if (left !== null) sub += left > 0 ? ' (' + left + ' araw pa)' : ' (expired na)';

        html += '<div class="item">';
        html += '<div style="min-width:0"><div class="t">' + esc(m.discord_id) + '</div>';
        html += '<div class="s"><span class="badge premium">' + esc(m.tier) + '</span> ' + esc(sub) + '</div></div>';
        html += '<button class="btn-danger" data-remove="' + esc(m.discord_id) + '">Alisin</button>';
        html += '</div>';
      });

      box.innerHTML = html;

      box.querySelectorAll('[data-remove]').forEach(function (btn) {
        btn.addEventListener('click', async function () {
          if (!confirm('Aalisin ang premium ng ' + btn.dataset.remove + '?')) return;
          try {
            await api(BASE + '/premium/' + encodeURIComponent(btn.dataset.remove), { method: 'DELETE' });
            toast('Naalis na.', 'ok');
            loadPremium();
          } catch (e) {
            toast(e.message, 'err');
          }
        });
      });
    } catch (e) {
      box.innerHTML = '<div class="empty">' + esc(e.message) + '</div>';
    }
  }

  document.getElementById('addPremium').addEventListener('click', async function () {
    var id = document.getElementById('pm_id').value.trim();
    var tier = document.getElementById('pm_tier').value.trim() || 'basic';
    var days = Number(document.getElementById('pm_days').value || 30);

    if (!/^[0-9]{15,25}$/.test(id)) {
      toast('Mali ang Discord ID. Numero lang iyon.', 'err');
      return;
    }

    try {
      await api(BASE + '/premium', { method: 'POST', body: { discord_id: id, tier: tier, days: days } });
      document.getElementById('pm_id').value = '';
      toast('Naidagdag. Ilang minuto bago lumabas ang role.', 'ok');
      loadPremium();
    } catch (e) {
      toast(e.message, 'err');
    }
  });

  // ---------- status board ----------
  async function loadStatus() {
    var box = document.getElementById('statusList');
    try {
      var out = await api(BASE + '/status');
      var list = out.entries || [];

      if (list.length === 0) {
        box.innerHTML = '<div class="empty">Wala pang entry.</div>';
        return;
      }

      var html = '';
      list.forEach(function (e) {
        html += '<div class="item">';
        html += '<div style="min-width:0"><div class="t">' + esc(e.title) + '</div>';
        html += '<div class="s"><span class="badge ' + esc(e.state) + '">' + esc(e.state) + '</span> ' +
                esc(e.game_name || '') + '</div></div>';
        html += '<button class="btn-danger" data-del="' + esc(e.id) + '">Burahin</button>';
        html += '</div>';
      });

      box.innerHTML = html;

      box.querySelectorAll('[data-del]').forEach(function (btn) {
        btn.addEventListener('click', async function () {
          try {
            await api(BASE + '/status/' + encodeURIComponent(btn.dataset.del), { method: 'DELETE' });
            toast('Nabura na.', 'ok');
            loadStatus();
          } catch (e) {
            toast(e.message, 'err');
          }
        });
      });
    } catch (e) {
      box.innerHTML = '<div class="empty">' + esc(e.message) + '</div>';
    }
  }

  document.getElementById('addStatus').addEventListener('click', async function () {
    var title = document.getElementById('st_title').value.trim();
    if (!title) {
      toast('Kailangan ng pamagat.', 'err');
      return;
    }

    var body = {
      title: title,
      game_name: document.getElementById('st_game').value.trim() || null,
      state: document.getElementById('st_state').value,
      note: document.getElementById('st_note').value.trim() || null
    };

    try {
      await api(BASE + '/status', { method: 'POST', body: body });
      document.getElementById('st_title').value = '';
      document.getElementById('st_game').value = '';
      document.getElementById('st_note').value = '';
      toast('Naidagdag sa board.', 'ok');
      loadStatus();
    } catch (e) {
      toast(e.message, 'err');
    }
  });

  // ---------- announce ----------
  document.getElementById('sendAnnounce').addEventListener('click', async function () {
    var title = document.getElementById('an_title').value.trim();
    var bodyText = document.getElementById('an_body').value.trim();

    if (!title || !bodyText) {
      toast('Kailangan ng pamagat at laman.', 'err');
      return;
    }

    var btn = this;
    btn.disabled = true;

    try {
      await api(BASE + '/announce', {
        method: 'POST',
        body: {
          title: title,
          body: bodyText,
          url: document.getElementById('an_url').value.trim() || null
        }
      });
      document.getElementById('an_title').value = '';
      document.getElementById('an_body').value = '';
      document.getElementById('an_url').value = '';
      toast('Nakapila na. Ilang segundo lang.', 'ok');
    } catch (e) {
      toast(e.message, 'err');
    }

    btn.disabled = false;
  });

  loadMappings();
  loadPremium();
  loadStatus();
})();
</script>
